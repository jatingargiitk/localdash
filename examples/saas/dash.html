Current MRR

 
DuckDB SQL
WITH a AS (SELECT greatest(max(signup_date), max(churned_at)) AS d FROM saas)
SELECT
  sum(mrr) FILTER (WHERE signup_date <= d AND (churned_at IS NULL OR churned_at > d)) AS mrr,
  sum(mrr) FILTER (WHERE signup_date <= d - INTERVAL 30 DAY AND (churned_at IS NULL OR churned_at > d - INTERVAL 30 DAY)) AS prev_mrr
FROM saas, a

Churn rate (30d)

 
DuckDB SQL
WITH a AS (SELECT greatest(max(signup_date), max(churned_at)) AS d FROM saas)
SELECT
  1.0 * count(*) FILTER (WHERE signup_date <= d - INTERVAL 30 DAY AND churned_at > d - INTERVAL 30 DAY AND churned_at <= d)
    / NULLIF(count(*) FILTER (WHERE signup_date <= d - INTERVAL 30 DAY AND (churned_at IS NULL OR churned_at > d - INTERVAL 30 DAY)), 0) AS churn_rate,
  1.0 * count(*) FILTER (WHERE signup_date <= d - INTERVAL 60 DAY AND churned_at > d - INTERVAL 60 DAY AND churned_at <= d - INTERVAL 30 DAY)
    / NULLIF(count(*) FILTER (WHERE signup_date <= d - INTERVAL 60 DAY AND (churned_at IS NULL OR churned_at > d - INTERVAL 60 DAY)), 0) AS prev_churn_rate
FROM saas, a

MRR by plan (month-end)

DuckDB SQL
WITH a AS (
  SELECT greatest(max(signup_date), max(churned_at)) AS d FROM saas
),
m AS (
  SELECT
    CAST(g AS DATE) AS month,
    least(CAST(g AS DATE) + INTERVAL 1 MONTH - INTERVAL 1 DAY, (SELECT d FROM a)) AS snap
  FROM (SELECT unnest(generate_series(date_trunc('month', min(signup_date)), date_trunc('month', max(signup_date)), INTERVAL 1 MONTH)) AS g FROM saas)
)
SELECT
  m.month AS month,
  s.plan AS plan,
  sum(s.mrr) AS mrr
FROM m
JOIN saas s
  ON s.signup_date <= m.snap
  AND (s.churned_at IS NULL OR s.churned_at > m.snap)
WHERE s.plan <> 'free'
GROUP BY 1, 2
ORDER BY 1

Monthly churn rate by plan

DuckDB SQL
WITH m AS (
  SELECT CAST(g AS DATE) AS month
  FROM (SELECT unnest(generate_series(date_trunc('month', min(signup_date)), date_trunc('month', max(signup_date)), INTERVAL 1 MONTH)) AS g FROM saas)
)
SELECT
  m.month AS month,
  s.plan AS plan,
  1.0 * count(*) FILTER (WHERE s.churned_at < m.month + INTERVAL 1 MONTH) / count(*) AS churn_rate
FROM m
JOIN saas s
  ON s.signup_date < m.month
  AND (s.churned_at IS NULL OR s.churned_at >= m.month)
GROUP BY 1, 2
ORDER BY 1
